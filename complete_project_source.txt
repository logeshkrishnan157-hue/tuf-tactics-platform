import os

# Output file where all project code will be combined
OUTPUT_FILE = "complete_project_source.txt"

# Folders or file extensions to ignore
IGNORE_DIRS = {
    "venv",
    ".git",
    "__pycache__",
    "node_modules",
    "data",
}
ALLOWED_EXTENSIONS = {".py", ".html", ".css", ".js", ".json"}


def bundle_project():
  current_dir = os.path.dirname(os.path.abspath(__file__))
  print(f"[INFO] Bundling project from: {current_dir}")

  total_files = 0

  with open(OUTPUT_FILE, "w", encoding="utf-8") as outfile:
    for root, dirs, files in os.walk(current_dir):
      # Modify dirs in-place to skip ignored directories
      dirs[:] = [d for d in dirs if d not in IGNORE_DIRS]

      for file in files:
        file_ext = os.path.splitext(file)[1].lower()

        # Skip output file itself and non-target extensions
        if file == OUTPUT_FILE or file_ext not in ALLOWED_EXTENSIONS:
          continue

        file_path = os.path.join(root, file)
        relative_path = os.path.relpath(file_path, current_dir)

        print(f"[ADDING] {relative_path}")

        outfile.write(f"\n{'='*80}\n")
        outfile.write(f"FILE: {relative_path}\n")
        outfile.write(f"{'='*80}\n\n")

        try:
          with open(file_path, "r", encoding="utf-8", errors="ignore") as infile:
            outfile.write(infile.read())
          outfile.write("\n\n")
          total_files += 1
        except Exception as e:
          outfile.write(f"[ERROR READING FILE: {e}]\n\n")

  print(
      f"\n[SUCCESS] Bundled {total_files} files successfully into"
      f" '{OUTPUT_FILE}'!"
  )


if __name__ == "__main__":
  bundle_project()

def extract_text_from_file(filepath):
    text = ""
    if filepath.endswith('.pdf'):
        try:
            reader = pypdf.PdfReader(filepath)
            for page in reader.pages:
                t = page.extract_text()
                if t: text += t + "\n"
        except Exception as e:
            print("PDF Extraction Error:", e)
    elif filepath.endswith('.docx'):
        try:
            doc = docx.Document(filepath)
            for para in doc.paragraphs:
                text += para.text + "\n"
        except Exception as e:
            print("DOCX Extraction Error:", e)
    return text.strip()

def is_valid_resume(text):
    lower_text = text.lower()
    resume_keywords = ['experience', 'skills', 'education', 'projects', 'summary', 'work', 'profile', 'developer', 'engineer', 'analyst', 'manager', 'degree', 'curriculum vitae', 'resume']
    matches = sum(1 for kw in resume_keywords if kw in lower_text)
    return len(text) > 80 and matches >= 2

def get_user_asked_questions(uid):
    past_results = InterviewResult.query.filter_by(user_id=uid).all()
    asked_texts = set()
    for res in past_results:
        if res.review_details:
            try:
                details = json.loads(res.review_details)
                for item in details:
                    if 'question' in item:
                        asked_texts.add(item['question'].strip())
            except:
                pass
    return asked_texts

@app.route('/mock-interview', methods=['GET', 'POST'])
def mock_interview():
    if 'user_id' not in session: return redirect(url_for('login'))
    uid = session.get('user_id')
    today_start = datetime.now().replace(hour=0, minute=0, second=0, microsecond=0)
    
    already_taken = InterviewResult.query.filter(InterviewResult.user_id == uid, InterviewResult.role.like("Interview:%"), InterviewResult.timestamp >= today_start).first()
    if already_taken:
        return render_template('mock_setup.html', error="⚠ Daily Limit Reached: You can only take 1 Mock Interview per day. Come back tomorrow!", theme=session.get('theme', 'dark'))

    if request.method == 'POST':
        target_role = request.form.get('target_role', 'Software Engineer').strip()
        resume_file = request.files.get('resume')
        
        if not resume_file or resume_file.filename == '':
            return render_template('mock_setup.html', error="⚠ Please upload a valid resume file (PDF or DOCX).", theme=session.get('theme', 'dark'))
        
        filename = secure_filename(resume_file.filename)
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        resume_file.save(filepath)
        
        resume_text = extract_text_from_file(filepath)
        if os.path.exists(filepath): os.remove(filepath)
        
        if not is_valid_resume(resume_text):
            return render_template('mock_setup.html', error="❌ Invalid Document! Uploaded file does not look like a professional resume. Please upload a valid resume containing your skills, experience, and education.", theme=session.get('theme', 'dark'))

        db.session.add(InterviewResult(
            user_id=uid, role=f"Interview: {target_role}", score=0, total=10, percentage=0,
            feedback_summary="Mock Interview In Progress / Session Started.", review_details=json.dumps([])
        ))
        db.session.commit()

        skill_keywords_count = sum(1 for word in ['python', 'java', 'sql', 'react', 'flask', 'aws', 'docker', 'machine learning', 'c++', 'javascript', 'html', 'css'] if word in resume_text.lower())
        q_count = 10 if skill_keywords_count < 3 else 20

        asked_texts = get_user_asked_questions(uid)
        cached_qs = QuestionCache.query.filter(QuestionCache.topic.ilike(f"%{target_role}%")).all()
        available_cached = [q for q in cached_qs if q.question_text.strip() not in asked_texts]
        
        questions = []
        if len(available_cached) >= q_count:
            selected_cached = random.sample(available_cached, q_count)
            for q in selected_cached:
                questions.append({
                    "id": len(questions)+1,
                    "category": q.category if q.category and "Core" not in q.category else f"{target_role} Core",
                    "question": q.question_text,
                    "options": json.loads(q.options),
                    "answer": q.answer,
                    "explanation": q.explanation
                })
        else:
            model = genai.GenerativeModel('gemini-3.8-flash')
            prompt = f"Based on resume, generate {q_count + 10} unique multi-round adaptive interview questions for '{target_role}' covering Technical, Situation Handling, and Aptitude rounds. Output strictly as a JSON array with keys: 'category' (PRECISE micro-topic or technical subfield like 'SAP MM - Vendor Consignment', 'SQL Joins', 'Python Functions', 'Inventory Valuation', etc. NEVER use generic terms like 'Core' or 'Skill Round'), 'question', 'options' (array of 4), 'answer', 'explanation'.\n\nRESUME CONTENT:\n{resume_text[:3000]}"
            try:
                response = model.generate_content(prompt)
                ai_data = json.loads(response.text.strip().replace("```json", "").replace("```", "").strip())
                
                fresh_items = []
                for item in ai_data:
                    q_text = item.get('question')
                    if q_text and q_text.strip() not in asked_texts:
                        fresh_items.append(item)
                
                for item in fresh_items[:q_count]:
                    opts = item.get('options')
                    ans = item.get('answer')
                    exp = item.get('explanation', "Professional expert rationale.")
                    q_text = item.get('question')
                    cat = item.get('category', f"{target_role} Core")
                    
                    existing_check = QuestionCache.query.filter_by(topic=target_role, question_text=q_text).first()
                    if not existing_check:
                        db.session.add(QuestionCache(topic=target_role, category=cat, question_text=q_text, options=json.dumps(opts), answer=ans, explanation=exp))
                    
                    questions.append({"id": len(questions)+1, "category": cat, "question": q_text, "options": opts, "answer": ans, "explanation": exp})
                db.session.commit()
            except Exception as e:
                return render_template('mock_setup.html', error="⚠ AI Generation Error: Failed to parse resume. Please try again.", theme=session.get('theme', 'dark'))

        active = ActiveInterview.query.filter_by(user_id=uid).first()
        if not active:
            active = ActiveInterview(user_id=uid, role=target_role, questions=json.dumps(questions), current_index=0, user_answers=json.dumps({}))
            db.session.add(active)
        else:
            active.role = target_role
            active.questions = json.dumps(questions)
            active.current_index = 0
            active.user_answers = json.dumps({})
        db.session.commit()

        return redirect(url_for('interview_live'))
    return render_template('mock_setup.html', theme=session.get('theme', 'dark'))

@app.route('/mock-live', methods=['GET', 'POST'])
def interview_live():
    if 'user_id' not in session: return redirect(url_for('login'))
    uid = session.get('user_id')
    active = ActiveInterview.query.filter_by(user_id=uid).first()
    if not active: return redirect(url_for('mock_interview'))

    questions = json.loads(active.questions)
    idx = active.current_index
    user_answers = json.loads(active.user_answers)

    if request.method == 'POST':
        user_answers[str(idx)] = request.form.get('answer')
        idx += 1
        active.current_index = idx
        active.user_answers = json.dumps(user_answers)
        db.session.commit()

        mistakes_count = 0
        for q_idx in range(idx):
            u_ans = user_answers.get(str(q_idx))
            if u_ans and str(u_ans).strip() != str(questions[q_idx]['answer']).strip():
                mistakes_count += 1

        if mistakes_count >= 5 or idx >= len(questions):
            role, score, detailed_review = active.role, 0, []
            weak_topics = []
            for q_idx in range(idx):
                q = questions[q_idx]
                user_ans = user_answers.get(str(q_idx), "Not Answered")
                is_correct = (str(user_ans).strip() == str(q['answer']).strip())
                if is_correct: score += 1
                else: weak_topics.append(q.get('category', role))
                detailed_review.append({"question": q['question'], "category": q['category'], "user_ans": user_ans, "correct_ans": q['answer'], "is_correct": is_correct, "explanation": q['explanation']})
            
            percentage = int((score / idx) * 100) if idx > 0 else 0
            feedback_msg = f"⚠ Early Exit Triggered (5 Mistakes reached)!" if mistakes_count >= 5 else f"🎉 Completed all questions successfully."
            
            res_record = InterviewResult.query.filter_by(user_id=uid, role=f"Interview: {role}").order_by(InterviewResult.id.desc()).first()
            if res_record:
                res_record.score = score
                res_record.total = idx
                res_record.percentage = percentage
                res_record.feedback_summary = f"{feedback_msg} Score: {score}/{idx} ({percentage}%). Focus area to practice: {', '.join(set(weak_topics))}"
                res_record.review_details = json.dumps(detailed_review)
            
            db.session.delete(active)
            db.session.commit()
            return redirect(url_for('dashboard'))

    return render_template('mock_live.html', question=questions[idx], index=idx+1, total=len(questions), role=active.role, theme=session.get('theme', 'dark'))

@app.route('/api/get-dynamic-quiz', methods=['GET'])
def get_dynamic_quiz():
    if 'user_id' not in session: return jsonify({"status": "unauthorized"}), 401
    uid = session.get('user_id')
    persona, subject, count = request.args.get('persona', 'Student'), request.args.get('subject', 'General').strip(), int(request.args.get('count', 10))
    
    asked_texts = get_user_asked_questions(uid)
    
    cached_questions = QuestionCache.query.filter_by(topic=subject).all()
    available_cached = [q for q in cached_questions if q.question_text.strip() not in asked_texts]
    
    quiz_list = []
    if len(available_cached) >= count:
        selected_qs = random.sample(available_cached, count)
        for q in selected_qs:
            quiz_list.append({"id": len(quiz_list)+1, "topic": subject, "category": q.category if q.category else subject, "question": q.question_text, "options": json.loads(q.options), "answer": q.answer, "explanation": q.explanation})
    else:
        model = genai.GenerativeModel('gemini-3.8-flash')
        response = model.generate_content(f"Generate {count + 10} unique multiple-choice questions on '{subject}' for a '{persona}'. Output strictly JSON array with keys: 'category' (SPECIFIC micro-topic or technical subfield, NEVER use generic names), 'question', 'options' (array of 4), 'answer', 'explanation'.")
        ai_data = json.loads(response.text.strip().replace("```json", "").replace("```", "").strip())
        
        fresh_items = []
        for item in ai_data:
            q_text = item.get('question')
            if q_text and q_text.strip() not in asked_texts:
                fresh_items.append(item)
                
        for item in fresh_items[:count]:
            q_text, opts, ans, exp = item.get('question'), item.get('options'), item.get('answer'), item.get('explanation', "Expert professional rationale.")
            cat = item.get('category', subject)
            
            existing_check = QuestionCache.query.filter_by(topic=subject, question_text=q_text).first()
            if not existing_check:
                db.session.add(QuestionCache(topic=subject, category=cat, question_text=q_text, options=json.dumps(opts), answer=ans, explanation=exp))
            
            quiz_list.append({"id": len(quiz_list)+1, "topic": subject, "category": cat, "question": q_text, "options": opts, "answer": ans, "explanation": exp})
        db.session.commit()
    return jsonify({"status": "success", "questions": quiz_list})

<script>
    let questions = []; let currentIndex = 0; let userAnswers = {}; let timer = 60; let timerInterval = null;
    
    async function loadQuiz() {
        let persona = "{{ persona }}"; let subject = "{{ subject }}"; let count = "{{ count }}";
        try {
            let res = await fetch(`/api/get-dynamic-quiz?persona=${encodeURIComponent(persona)}&subject=${encodeURIComponent(subject)}&count=${count}`);
            let data = await res.json();
            if(data.status === 'success' && data.questions.length > 0) {
                questions = data.questions;
                currentIndex = 0; userAnswers = {}; showQuestion();
            } else { document.getElementById('q-title').innerText = "Failed to generate questions."; }

    }

    function startTimer() {
        timer = 60; document.getElementById('time-left').innerText = timer; document.getElementById('hint-btn').classList.add('hidden');
        if(timerInterval) clearInterval(timerInterval);
        timerInterval = setInterval(() => {
            timer--; if(timer > 0) { document.getElementById('time-left').innerText = timer; }
            else { clearInterval(timerInterval); document.getElementById('time-left').innerText = "0"; document.getElementById('hint-btn').classList.remove('hidden'); }
        }, 1000);

        let q = questions[currentIndex]; let buttons = document.querySelectorAll('.option-btn'); let hiddenCount = 0;

        document.getElementById('hint-btn').classList.add('hidden');
    }

def extract_text_from_file(filepath):
    text = ""
    if filepath.endswith('.pdf'):
        try:
            reader = pypdf.PdfReader(filepath)
            for page in reader.pages:
                t = page.extract_text()
                if t: text += t + "\\n"
        except Exception as e:
            print("PDF Extraction Error:", e)
    elif filepath.endswith('.docx'):
        try:
            doc = docx.Document(filepath)
            for para in doc.paragraphs:
                text += para.text + "\\n"
        except Exception as e:
            print("DOCX Extraction Error:", e)
    return text.strip()

    matches = sum(1 for kw in resume_keywords if kw in lower_text)
    return len(text) > 80 and matches >= 2

def mock_interview():

    uid = session.get('user_id')

    if already_taken:

        asked_texts = get_user_asked_questions(uid)
        cached_qs = QuestionCache.query.filter(QuestionCache.topic.ilike(f"%{target_role}%")).all()
        available_cached = [q for q in cached_qs if q.question_text.strip() not in asked_texts]
        
        questions = []
        if len(available_cached) >= q_count:
            selected_cached = random.sample(available_cached, q_count)
            for q in selected_cached:
                questions.append({
                    "id": len(questions)+1,
                    "category": q.category if q.category and "Core" not in q.category else f"{target_role} Core",
                    "question": q.question_text,
                    "options": json.loads(q.options),
                    "answer": q.answer,
                    "explanation": q.explanation
                })
        else:
            model = genai.GenerativeModel('gemini-3.8-flash')
            prompt = f"Based on resume, generate {q_count + 10} unique multi-round adaptive interview questions for '{target_role}' covering Technical, Situation Handling, and Aptitude rounds. Output strictly as a JSON array with keys: 'category' (PRECISE micro-topic or technical subfield like 'SAP MM - Vendor Consignment', 'SQL Joins', 'Python Functions', 'Inventory Valuation', etc. NEVER use generic terms like 'Core' or 'Skill Round'), 'question', 'options' (array of 4), 'answer', 'explanation'.\\n\\nRESUME CONTENT:\\n{resume_text[:3000]}"
            try:
                response = model.generate_content(prompt)
                ai_data = json.loads(response.text.strip().replace("```json", "").replace("```", "").strip())

                for item in ai_data:
                    q_text = item.get('question')
                    if q_text and q_text.strip() not in asked_texts:
                        fresh_items.append(item)

                    questions.append({"id": len(questions)+1, "category": cat, "question": q_text, "options": opts, "answer": ans, "explanation": exp})
                db.session.commit()
            except Exception as e:
                return render_template('mock_setup.html', error="⚠ AI Generation Error: Failed to parse resume. Please try again.", theme=session.get('theme', 'dark'))

        active = ActiveInterview.query.filter_by(user_id=uid).first()

@app.route('/mock-live', methods=['GET', 'POST'])
def interview_live():
    if 'user_id' not in session: return redirect(url_for('login'))
    uid = session.get('user_id')
    active = ActiveInterview.query.filter_by(user_id=uid).first()
    if not active: return redirect(url_for('mock_interview'))

    questions = json.loads(active.questions)
    idx = active.current_index
    user_answers = json.loads(active.user_answers)

    if request.method == 'POST':
        user_answers[str(idx)] = request.form.get('answer')
        idx += 1
        active.current_index = idx
        active.user_answers = json.dumps(user_answers)
        db.session.commit()

        mistakes_count = 0
        for q_idx in range(idx):
            u_ans = user_answers.get(str(q_idx))
            if u_ans and str(u_ans).strip() != str(questions[q_idx]['answer']).strip():
                mistakes_count += 1

        if mistakes_count >= 5 or idx >= len(questions):
            role, score, detailed_review = active.role, 0, []
            weak_topics = []
            for q_idx in range(idx):
                q = questions[q_idx]
                user_ans = user_answers.get(str(q_idx), "Not Answered")
                is_correct = (str(user_ans).strip() == str(q['answer']).strip())
                if is_correct: score += 1
                else: weak_topics.append(q.get('category', role))
                detailed_review.append({"question": q['question'], "category": q['category'], "user_ans": user_ans, "correct_ans": q['answer'], "is_correct": is_correct, "explanation": q['explanation']})
            
            percentage = int((score / idx) * 100) if idx > 0 else 0
            feedback_msg = f"⚠ Early Exit Triggered (5 Mistakes reached)!" if mistakes_count >= 5 else f"🎉 Completed all questions successfully."
            
            res_record = InterviewResult.query.filter_by(user_id=uid, role=f"Interview: {role}").order_by(InterviewResult.id.desc()).first()
            if res_record:
                res_record.score = score
                res_record.total = idx
                res_record.percentage = percentage
                res_record.feedback_summary = f"{feedback_msg} Score: {score}/{idx} ({percentage}%). Focus area to practice: {', '.join(set(weak_topics))}"
                res_record.review_details = json.dumps(detailed_review)
            
            db.session.delete(active)
            db.session.commit()
            return redirect(url_for('dashboard'))

    return render_template('mock_live.html', question=questions[idx], index=idx+1, total=len(questions), role=active.role, theme=session.get('theme', 'dark'))

<script>
    let questions = []; let currentIndex = 0; let userAnswers = {}; let timer = 60; let timerInterval = null;
    
    async function loadQuiz() {
        let persona = "{{ persona }}"; let subject = "{{ subject }}"; let count = "{{ count }}";
        try {
            let res = await fetch(`/api/get-dynamic-quiz?persona=${encodeURIComponent(persona)}&subject=${encodeURIComponent(subject)}&count=${count}`);
            let data = await res.json();
            if(data.status === 'success' && data.questions.length > 0) {
                questions = data.questions;
                currentIndex = 0; userAnswers = {}; showQuestion();
            } else { document.getElementById('q-title').innerText = "Failed to generate questions."; }
        } catch (err) { console.error("Error loading quiz:", err); document.getElementById('q-title').innerText = "Error connecting to server."; }
    }

    function startTimer() {
        timer = 60; document.getElementById('time-left').innerText = timer; document.getElementById('hint-btn').classList.add('hidden');
        if(timerInterval) clearInterval(timerInterval);
        timerInterval = setInterval(() => {
            timer--; if(timer > 0) { document.getElementById('time-left').innerText = timer; }
            else { clearInterval(timerInterval); document.getElementById('time-left').innerText = "0"; document.getElementById('hint-btn').classList.remove('hidden'); }
        }, 1000);
    }
    function useHint() {
        let q = questions[currentIndex]; let buttons = document.querySelectorAll('.option-btn'); let hiddenCount = 0;
        buttons.forEach((btn) => { if(btn.getAttribute('data-opt') !== q.answer && hiddenCount < 2) { btn.style.opacity = '0.2'; btn.style.pointerEvents = 'none'; hiddenCount++; } });
        document.getElementById('hint-btn').classList.add('hidden');
    }
    function showQuestion() {
        if(currentIndex >= questions.length) { if(timerInterval) clearInterval(timerInterval); finishQuiz(); return; }
        startTimer(); document.getElementById('progress-indicator').innerText = `Question ${currentIndex + 1} of ${questions.length}`;
        let q = questions[currentIndex]; document.getElementById('q-title').innerText = `${currentIndex + 1}. ${q.question}`;
        let optBox = document.getElementById('options-box'); optBox.innerHTML = '';
        q.options.forEach((opt, idx) => {
            let isSelected = userAnswers[currentIndex] === opt;
            let activeClass = isSelected ? 'bg-indigo-600/20 border-indigo-500 font-bold shadow-md' : 'glass-card hover:border-indigo-500/40';
            optBox.innerHTML += `<button onclick="selectOption('${escapeHtml(opt)}', this)" data-opt="${escapeHtml(opt)}" class="option-btn w-full text-left p-4 rounded-2xl border transition font-medium text-sm sm:text-base ${activeClass}"><span class="font-bold text-indigo-600 dark:text-indigo-400 mr-2">${String.fromCharCode(65 + idx)}.</span> <span>${opt}</span></button>`;
        });
        let nextBtn = document.getElementById('next-btn');
        if(userAnswers[currentIndex] !== undefined) nextBtn.classList.remove('hidden'); else nextBtn.classList.add('hidden');
    }
    function selectOption(optVal, btnElement) {
        userAnswers[currentIndex] = optVal;
        document.querySelectorAll('.option-btn').forEach(btn => { btn.classList.remove('bg-indigo-600/20', 'border-indigo-500', 'font-bold', 'shadow-md'); btn.classList.add('glass-card'); });
        btnElement.classList.remove('glass-card'); btnElement.classList.add('bg-indigo-600/20', 'border-indigo-500', 'font-bold', 'shadow-md');
        document.getElementById('next-btn').classList.remove('hidden');
    }
    function nextQuestion() { currentIndex++; showQuestion(); }
    async function finishQuiz() {
        document.getElementById('question-box').classList.add('hidden'); document.getElementById('result-box').classList.remove('hidden');
        let score = 0; let reviewList = [];
        questions.forEach((q, idx) => {
            let userAns = userAnswers[idx] || "Not Answered";
            let isCorrect = (String(userAns).trim() === String(q.answer).trim());
            if(isCorrect) score++;
            reviewList.push({ "category": q.category || "Mock Test", "question": q.question, "user_ans": userAns, "correct_ans": q.answer, "is_correct": isCorrect, "explanation": q.explanation });
        });
        let percentage = Math.round((score / questions.length) * 100);
        try { await fetch('/api/save-quiz-result', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ subject: "{{ subject }}", score, total: questions.length, percentage, review: reviewList }) }); } catch(e) { console.error("Error saving result:", e); }
        setTimeout(() => { window.location.href = "/dashboard"; }, 1200);
    }
    function escapeHtml(text) { return text.replace(/'/g, "&#39;").replace(/"/g, "&quot;"); }
    loadQuiz();
</script>

import os
import json
import google.generativeai as genai
from flask import Blueprint, jsonify, request
from app.models import db, QuestionCache

gemini_bp = Blueprint('gemini_bp', __name__)

def call_gemini_working_key(prompt):
    api_key = os.getenv("GEMINI_API_KEY")
    if not api_key:
        return None

    try:
        genai.configure(api_key=api_key)
        
        # Using the exact recommended model gemini-3.8-flash
        model = genai.GenerativeModel('gemini-3.8-flash')
        response = model.generate_content(prompt)
        return response.text
    except Exception as e:
        print(f"Error with 3.8-flash: {e}")
        try:
            # Fallback to general flash variant if needed
            model = genai.GenerativeModel('gemini-flash')
            response = model.generate_content(prompt)
            return response.text
        except Exception as e2:
            print(f"Fallback Error: {e2}")
            return None

@gemini_bp.route('/generate-unique-questions', methods=['POST'])
def generate_unique_questions():
    data = request.json
    topic = data.get('topic', 'General Programming')
    count = int(data.get('count', 10))

    existing = QuestionCache.query.filter_by(topic=topic).all()
    existing_texts = [q.question_text for q in existing]

    prompt = f"""
    Generate {count} unique multiple-choice interview questions for the topic: {topic}.
    Ensure these questions are completely distinct from these already asked questions: {existing_texts[-30:]}
    Return strictly as a JSON list of objects with keys: "question", "options" (list of 4 options), "answer", "explanation". No markdown formatting outside json if possible, just raw JSON.
    """

    raw_response = call_gemini_working_key(prompt)
    if not raw_response:
        return jsonify({"error": "API key failed to generate response!"}), 500

    try:
        cleaned_json = raw_response.replace("```json", "").replace("```", "").strip()
        questions_list = json.loads(cleaned_json)

        new_saved_questions = []
        for q in questions_list:
            q_text = q.get('question')
            if not QuestionCache.query.filter_by(question_text=q_text).first():
                opts = q.get('options', ["A", "B", "C", "D"])
                new_q = QuestionCache(
                    topic=topic,
                    question_text=q_text,
                    option_a=opts[0] if len(opts) > 0 else "",
                    option_b=opts[1] if len(opts) > 1 else "",
                    option_c=opts[2] if len(opts) > 2 else "",
                    option_d=opts[3] if len(opts) > 3 else "",
                    correct_answer=q.get('answer'),
                    explanation=q.get('explanation')
                )
                db.session.add(new_q)
                new_saved_questions.append(q)

        db.session.commit()
        return jsonify({"status": "success", "new_questions": new_saved_questions})
    
    except Exception as e:
        return jsonify({"error": str(e), "raw": raw_response}), 400

@gemini_bp.route('/test-ai')
def test_ai():
    prompt = "Give a 1-sentence welcoming quote for a smart EdTech platform students."
    ai_output = call_gemini_working_key(prompt)
    return jsonify({"response": ai_output})
